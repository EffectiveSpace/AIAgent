class AdvancedProtocol(Protocol):
    def calculate_crc_sum(self, data: bytes) -> int:
        """Сумма всех байтов по модулю 256"""
        total = sum(data) % 256
        return total
    
    def calculate_crc_positional(self, data: bytes) -> int:
        """Сумма с умножением на позицию"""
        total = 0
        for i, byte in enumerate(data):

### Пример оптимизации
```python
class OptimizedArduinoInterface(ArduinoProtocolInterface):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.state_cache = {}  # Кэш состояний устройств
        self.command_batch = []  # Буфер для команд
    
    def read_sensor_cached(self, sensor_id: int, ttl: float = 1.0):
        """Чтение сенсора с кэшированием"""
        current_time = time.time()
        cache_key = f"sensor_{sensor_id}"
        
        if cache_key in self.state_cache:
            cached_time, cached_value = self.state_cache[cache_key]
            if current_time - cached_time < ttl:
                return cached_value
        
        # Запрос к устройству
        result = self.send_command(0x04, [sensor_id])
        self.state_cache[cache_key] = (current_time, result)
        return result
```
